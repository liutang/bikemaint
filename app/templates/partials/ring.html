{# Progress ring for an urgent task. Meaning is always in text (aria-label + visible value), never color alone. #} {% macro ring(item) -%} {%- set d = item.days_until -%} {%- if item.status == "overdue" -%} {%- if d is none or d >= 0 -%} {%- set value, caption, label = "Late", "", "Overdue" -%} {%- else -%} {%- set value, caption, label = (-d ~ "d"), "late", ("Overdue by " ~ (-d) ~ (" day" if d == -1 else " days")) -%} {%- endif -%} {%- elif d is none or d <= 0 -%} {%- set value, caption, label = "Today", "", "Due today" -%} {%- else -%} {%- set value, caption, label = (d ~ "d"), "left", ("Due in " ~ d ~ (" day" if d == 1 else " days")) -%} {%- endif -%}
{{ value }}{% if caption %}{{ caption }}{% endif %}
{%- endmacro %}
